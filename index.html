    <script>

        // Database State

        let db = JSON.parse(localStorage.getItem('ecosort_ai_db')) || {

            students: [

                { id: "S001", name: "Jude Louise Loria", room: "Grade 11 Heisenberg", points: 40 },                 { id: "S002", name: "Maria Santos", room: "Grade 11 Heisenberg", points: 25 }

            ],

            disposals: []

        };

let currentRole = "admin"; // "admin" or "student" let activeStudent = db.students[0];

let activeBin = null;

let detectedItemLabel = "";

let cocoModel = null;

        // Rule Map: Maps COCO-SSD object classes to school bin types         const ITEM_BIN_RULES = {

            "bottle": "Recyclable",

            "cup": "Recyclable",

            "cell phone": "Recyclable",

            "book": "Recyclable",

            "apple": "Biodegradable",

            "banana": "Biodegradable",

            "sandwich": "Biodegradable"

        };

        function saveDB() {

            localStorage.setItem('ecosort_ai_db', JSON.stringify(db));             updateUI();

        }

        // Camera Permission Request

        async function requestCameraPermission() {

            const video = document.getElementById('webcam');

            const prompt = document.getElementById('camera-prompt');             const statusBadge = document.getElementById('ai-status');

            try {

                const stream = await navigator.mediaDevices.getUserMedia({                      video: { facingMode: "environment" } 

                });

                video.srcObject = stream;

                video.classList.remove('hidden');

                prompt.classList.add('hidden');

                statusBadge.innerText = "Loading AI Model...";

                statusBadge.className = "text-xs bg-blue-100 text-blue-800 px-2 py-0.5 rounded-full font-semibold";

                // Load COCO-SSD Model (Mobile-friendly YOLO alternative)

                cocoModel = await cocoSsd.load();

                statusBadge.innerText = "AI Live";

                statusBadge.className = "text-xs bg-emerald-100 text-emerald-800 px-2 py-0.5 rounded-full font-semibold";

                detectFrame();

            } catch (err) {

                alert("Camera Permission Denied: Please allow camera access in your browser settings to run object detection.");

                console.error(err);

            }

        }

        // Continuous Object Detection Loop

        async function detectFrame() {

            const video = document.getElementById('webcam');             if (cocoModel && video.readyState === 4) {

                const predictions = await cocoModel.detect(video);

                if (predictions.length > 0) {

                    detectedItemLabel = predictions[0].class;

                    document.getElementById('predicted-item').innerText = detectedItemLabel.toUpperCase();

                    checkLogEligibility();

                } else {

                    detectedItemLabel = "";

                    document.getElementById('predicted-item').innerText = "Scanning...";                     disableLogButton();

                }

            }

            requestAnimationFrame(detectFrame);

        }

        function checkLogEligibility() {

            const logBtn = document.getElementById('log-btn');

            if (activeStudent && activeBin && detectedItemLabel) {

                logBtn.disabled = false;

                logBtn.classList.remove('opacity-50', 'cursor-not-allowed');             }

        }

        function disableLogButton() {

            const logBtn = document.getElementById('log-btn');

            logBtn.disabled = true;

            logBtn.classList.add('opacity-50', 'cursor-not-allowed');         }

        // Session Setup Handlers

        function selectStudent() {

            let list = db.students.map((s, i) => `${i+1}. ${s.name} (${s.room})`).join('\n');

            let choice = prompt("Select Student:\n" + list);

            let idx = parseInt(choice) - 1;

            if (!isNaN(idx) && db.students[idx]) {

                activeStudent = db.students[idx];

                document.getElementById('active-user-text').innerText = activeStudent.name;                 checkLogEligibility();

            }

        }

        function selectBin() {

            let choice = prompt("Select Bin Type:\n1. Biodegradable\n2. Recyclable\n3. Residual");             if (choice === "1") activeBin = "Biodegradable";

            else if (choice === "2") activeBin = "Recyclable";

            else if (choice === "3") activeBin = "Residual";

            if (activeBin) {

                document.getElementById('active-bin-text').innerText = activeBin;                 checkLogEligibility();

            }

}

        // Log AI Disposal Transaction

        function logAIDisposal() {

            if (!activeStudent || !activeBin || !detectedItemLabel) return;

let expectedBin = ITEM_BIN_RULES[detectedItemLabel] || "Residual"; let isProper = (activeBin === expectedBin);

let pointsEarned = isProper ? 10 : 0;

            db.disposals.push({

                studentId: activeStudent.id,

                studentName: activeStudent.name,

                bin: activeBin,

                item: detectedItemLabel,

                isProper: isProper,

                timestamp: new Date().toLocaleTimeString()             });

            if (isProper) {

                let student = db.students.find(s => s.id === activeStudent.id);

                if (student) student.points += pointsEarned;

                alert(`🎉 SUCCESS! Item '${detectedItemLabel}' properly placed in ${activeBin} Bin. +10 Eco-Points!`);

            } else {

                alert(`❌ MIS-SORTED! '${detectedItemLabel}' belongs in the ${expectedBin} Bin, not ${activeBin}. 0 points awarded.`);

            }

            saveDB();

        }

        // Role & Tab Navigation

        function toggleRole() {

            currentRole = (currentRole === "admin") ? "student" : "admin";

            document.getElementById('role-btn').innerText = `Role: ${currentRole === "admin" ? "Admin" : "Student"}`;

const dash = document.getElementById('student-dashboard'); const adminNav = document.getElementById('nav-rooms');

            if (currentRole === "student") {

                dash.classList.remove('hidden');

                adminNav.classList.add('hidden');

            } else {

                dash.classList.add('hidden');

                adminNav.classList.remove('hidden');

            }

        }

        function switchTab(tab) {

            ['scan', 'stats', 'rooms'].forEach(t => {

                document.getElementById(`tab-${t}`).classList.add('hidden');

                document.getElementById(`nav-${t}`).className = "text-slate-400 font-semibold space-y-0.5";

            });

            document.getElementById(`tab-${tab}`).classList.remove('hidden');

            document.getElementById(`nav-${tab}`).className = "text-emerald-600 font-bold space-y-0.5";

            if (tab === 'stats') renderStats();

        }

        // Admin Actions

        function addStudent() {

            let name = document.getElementById('new-student-name').value.trim();             let room = document.getElementById('new-student-room').value;

            if (name && room) {

                let id = "S0" + (db.students.length + 1);

                db.students.push({ id, name, room, points: 0 });

                document.getElementById('new-student-name').value = "";

                saveDB();

            }

        }

        function renderStats() {

            let total = db.disposals.length;

            document.getElementById('stat-total').innerText = total;

            if (total === 0) {

                document.getElementById('stat-accuracy').innerText = "0%";

                document.getElementById('bin-stats-list').innerHTML = `<p class="text-slate-400 text-center py-2">No records logged yet.</p>`;

                return;

            }

            let proper = db.disposals.filter(d => d.isProper).length;

            document.getElementById('stat-accuracy').innerText = Math.round((proper / total) * 100) + "%";

            let bins = ["Biodegradable", "Recyclable", "Residual"];

            let listHTML = bins.map(b => {

                let items = db.disposals.filter(d => d.bin === b);

                let correct = items.filter(d => d.isProper).length;

                let rate = items.length > 0 ? Math.round((correct / items.length) * 100) : 0;

                return `

                    <div class="flex justify-between items-center bg-slate-50 p-2.5 rounded-xl border border-slate-100">

                        <span class="font-semibold text-slate-700">${b} Bin</span>

                        <span class="font-bold text-emerald-600">${rate}% Proper (${items.length} items)</span>

                    </div>

                `;

            }).join('');

            document.getElementById('bin-stats-list').innerHTML = listHTML;

        }

        function updateUI() {

            // Update Student Ledger

            let ledgerHTML = db.students.map(s => `

                <div class="flex justify-between items-center bg-slate-50 p-2.5 rounded-xl border border-slate-100">

                    <div>

                        <span class="font-semibold text-slate-700">${s.name}</span>

                        <span class="block text-slate-400 text-2xs">${s.room}</span>

                    </div>

                    <span class="bg-emerald-100 text-emerald-800 font-bold px-2 py-0.5 rounded-md">${s.points} pts</span>

                </div>

            `).join('');

            document.getElementById('students-ledger').innerHTML = ledgerHTML;

            // Update Active Student UI Displays

            if (activeStudent) {

                document.getElementById('student-name-display').innerText = activeStudent.name;                 document.getElementById('student-room-display').innerText = activeStudent.room;

                document.getElementById('student-points-display').innerText = activeStudent.points;             }

        }

        // Initial Setup

        updateUI();
